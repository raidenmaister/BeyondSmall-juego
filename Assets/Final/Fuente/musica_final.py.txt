"""Musica original para el final de Beyond Small (sintetizada con numpy).
Final_Musica.wav: ~2:25, re menor -> re mayor. Intro tierna (campanitas) -> crece (cuerdas, arpegios)
-> epica (tambores, metales, coro) -> resolucion luminosa.
Final_Crecer.wav: subida de 7 s (barrido + glissando + brillo) que termina en un golpe grave.
"""
import numpy as np, sys, wave, os

SR = 44100
OUT = sys.argv[1] if len(sys.argv) > 1 else "."
rng = np.random.default_rng(7)

BPM = 84.0
BEAT = 60.0 / BPM
BAR = 4 * BEAT

def nota(n):  # nombre MIDI -> Hz
    return 440.0 * 2 ** ((n - 69) / 12.0)

def env_adsr(n, a, d, s, r, total):
    t = np.arange(n) / SR
    e = np.ones(n) * s
    e[t < a] = t[t < a] / max(a, 1e-4)
    m = (t >= a) & (t < a + d)
    e[m] = 1 - (1 - s) * (t[m] - a) / max(d, 1e-4)
    rel = t > total - r
    e[rel] *= np.clip((total - t[rel]) / max(r, 1e-4), 0, 1)
    return e

def lp(x, fc):
    """filtro pasa bajos de un polo (fc puede ser arreglo)"""
    fc = np.broadcast_to(np.asarray(fc, float), x.shape)
    a = np.exp(-2 * np.pi * fc / SR)
    y = np.empty_like(x); z = 0.0
    # vectorizado por bloques no es trivial; usamos un bucle numba-free pero en trozos con escala fija
    for i in range(len(x)):
        z = (1 - a[i]) * x[i] + a[i] * z
        y[i] = z
    return y

def lp_fijo(x, fc):
    # pasa bajos rapido por FFT (respuesta suave)
    n = len(x); N = 1 << int(np.ceil(np.log2(n + 1)))
    X = np.fft.rfft(x, N)
    f = np.fft.rfftfreq(N, 1 / SR)
    H = 1 / np.sqrt(1 + (f / fc) ** 4)
    return np.fft.irfft(X * H, N)[:n]

def hp_fijo(x, fc):
    n = len(x); N = 1 << int(np.ceil(np.log2(n + 1)))
    X = np.fft.rfft(x, N)
    f = np.fft.rfftfreq(N, 1 / SR)
    H = 1 - 1 / np.sqrt(1 + (f / fc) ** 4)
    return np.fft.irfft(X * H, N)[:n]

def saw(f, n, fase=0.0):
    t = np.arange(n) / SR
    ph = (f * t + fase) % 1.0
    return 2 * ph - 1

def sine(f, n, fase=0.0):
    return np.sin(2 * np.pi * (f * np.arange(n) / SR + fase))

class Pista:
    def __init__(self, dur):
        self.n = int(dur * SR)
        self.L = np.zeros(self.n); self.R = np.zeros(self.n)
    def add(self, t0, x, pan=0.0, vol=1.0):
        i0 = int(t0 * SR)
        if i0 >= self.n: return
        x = x[: self.n - i0] * vol
        gl = np.cos((pan + 1) * np.pi / 4); gr = np.sin((pan + 1) * np.pi / 4)
        self.L[i0:i0 + len(x)] += x * gl
        self.R[i0:i0 + len(x)] += x * gr

# ------------------------------------------------------------------ instrumentos
def pad(f, dur, brillo=1400, vol=0.12):
    n = int(dur * SR)
    x = sum(saw(f * 2 ** (c / 1200), n, rng.random()) for c in (-7, 0, 6))
    x = lp_fijo(x, brillo)
    vib = 1 + 0.002 * sine(4.8, n)
    return x * env_adsr(n, 1.6, 0.5, 0.85, 1.6, dur) * vol * vib

def coro(f, dur, vol=0.07):
    n = int(dur * SR)
    t = np.arange(n) / SR
    vib = 5.0 * np.sin(2 * np.pi * 5.2 * t)
    x = np.zeros(n)
    for k, amp in ((1, 1.0), (2, 0.5), (3, 0.35), (4, 0.12), (5, 0.08)):
        x += amp * np.sin(2 * np.pi * (f * k * t + np.cumsum(vib * k) / SR * 0.003))
    x = lp_fijo(x, 2200)
    return x * env_adsr(n, 2.0, 0.5, 0.9, 2.0, dur) * vol

def campana(f, dur=2.6, vol=0.22):
    n = int(dur * SR)
    t = np.arange(n) / SR
    x = (np.sin(2 * np.pi * f * t) * np.exp(-t * 1.6) + 0.45 * np.sin(2 * np.pi * f * 2.0 * t) * np.exp(-t * 2.6)
         + 0.25 * np.sin(2 * np.pi * f * 3.01 * t) * np.exp(-t * 4) + 0.12 * np.sin(2 * np.pi * f * 4.2 * t) * np.exp(-t * 6))
    ataque = np.clip(t / 0.004, 0, 1)
    return x * ataque * vol

def cuerda_corta(f, dur, vol=0.09):
    n = int(dur * SR)
    x = saw(f, n) + 0.6 * saw(f * 1.003, n)
    x = lp_fijo(x, 2600)
    t = np.arange(n) / SR
    return x * np.clip(t / 0.02, 0, 1) * np.exp(-t * 3.2) * vol

def metal(f, dur, vol=0.1):
    n = int(dur * SR)
    t = np.arange(n) / SR
    x = saw(f, n) + saw(f * 1.004, n, 0.3) + 0.5 * saw(f * 0.996, n, 0.6)
    # el filtro se abre con el ataque (sonido de metales)
    a = lp_fijo(x, 700); b = lp_fijo(x, 3200)
    k = np.clip(t / 0.35, 0, 1) * np.exp(-t * 0.25)
    y = a * (1 - k) + b * k
    return y * env_adsr(n, 0.12, 0.3, 0.8, 0.6, dur) * vol

def bajo(f, dur, vol=0.18):
    n = int(dur * SR)
    x = sine(f, n) + 0.35 * lp_fijo(saw(f, n), 400)
    return x * env_adsr(n, 0.05, 0.2, 0.8, 0.4, dur) * vol

def timbal(vol=0.5, f0=62):
    dur = 1.6; n = int(dur * SR); t = np.arange(n) / SR
    f = f0 * (1 + 0.6 * np.exp(-t * 18))
    x = np.sin(2 * np.pi * np.cumsum(f) / SR) * np.exp(-t * 2.6)
    ruido = lp_fijo(rng.standard_normal(n), 900) * np.exp(-t * 30) * 0.6
    return (x + ruido) * vol

def taiko(vol=0.6):
    dur = 0.9; n = int(dur * SR); t = np.arange(n) / SR
    f = 52 * (1 + 1.2 * np.exp(-t * 40))
    x = np.sin(2 * np.pi * np.cumsum(f) / SR) * np.exp(-t * 5)
    click = hp_fijo(rng.standard_normal(n), 2000) * np.exp(-t * 120) * 0.25
    return (x + click) * vol

def platillo(dur=3.0, vol=0.08, crece=True):
    n = int(dur * SR); t = np.arange(n) / SR
    x = hp_fijo(rng.standard_normal(n), 5000)
    e = (t / dur) ** 2 if crece else np.exp(-t * 1.2)
    return x * e * vol

# ------------------------------------------------------------------ composicion
# acordes (raiz MIDI y tipo) en re menor: Dm Bb F C | Gm Bb Dm A ...
def triada(raiz, menor):
    return [raiz, raiz + (3 if menor else 4), raiz + 7]

DM, BB, F_, C_, GM, A_ = (triada(50, True), triada(46, False), triada(53, False), triada(48, False), triada(43, True), triada(45, False))
D_ = triada(50, False); G_ = triada(43, False)
prog_a = [DM, BB, F_, C_]
prog_b = [GM, BB, DM, A_]

secciones = [("intro", 6), ("crece", 12), ("epica", 22), ("final", 8)]
total_bars = sum(b for _, b in secciones)
DUR = total_bars * BAR + 9.0
P = Pista(DUR)

# melodia principal (en notas MIDI, duraciones en pulsos)
tema = [(69, 1), (74, 1), (76, 1), (77, 1), (76, 2), (74, 1), (72, 1),
        (70, 1), (72, 1), (74, 1), (77, 1), (76, 3), (None, 1),
        (74, 1), (76, 1), (77, 1), (79, 1), (81, 2), (79, 1), (77, 1),
        (76, 1), (74, 1), (72, 1), (73, 1), (74, 4)]

def tocar_melodia(t0, instrumento, transp=0, vol=1.0, pan=0.0):
    t = t0
    for (m, b) in tema:
        if m is not None:
            P.add(t, instrumento(nota(m + transp), b * BEAT * 1.4) * vol, pan)
        t += b * BEAT
    return t

bar = 0
for nombre, nb in secciones:
    for k in range(nb):
        t = bar * BAR
        if nombre == "final":
            acorde = [BB, C_, D_, D_][min(k // 2, 3)]
        else:
            acorde = (prog_a if (bar // 4) % 2 == 0 else prog_b)[bar % 4]
        raiz = acorde[0]
        # pad siempre (mas brillante cuanto mas avanza)
        brillo = {"intro": 900, "crece": 1500, "epica": 2600, "final": 1800}[nombre]
        vpad = {"intro": 0.06, "crece": 0.09, "epica": 0.11, "final": 0.1}[nombre]
        for i, m in enumerate(acorde + [acorde[0] + 12]):
            P.add(t, pad(nota(m + 12), BAR + 1.6, brillo, vpad), pan=(-0.5 + i * 0.33))
        # intro: campanitas en arpegio lento
        if nombre in ("intro", "final"):
            for j, m in enumerate([acorde[0] + 24, acorde[1] + 24, acorde[2] + 24, acorde[1] + 36]):
                P.add(t + j * BEAT, campana(nota(m), 3.0, 0.11 if nombre == "intro" else 0.09), pan=(-0.4 + 0.25 * j))
        # crece: arpegio de cuerdas en corcheas y bajo
        if nombre in ("crece", "epica"):
            notas = [acorde[0] + 12, acorde[1] + 12, acorde[2] + 12, acorde[1] + 24]
            for j in range(8):
                P.add(t + j * BEAT / 2, cuerda_corta(nota(notas[j % 4] + 12), 0.6, 0.06 if nombre == "crece" else 0.075), pan=(0.35 if j % 2 else -0.35))
            P.add(t, bajo(nota(raiz - 12), BAR * 0.95, 0.11))
        # epica: metales, coro, tambores
        if nombre == "epica":
            for m in acorde:
                P.add(t, metal(nota(m), BAR * 0.98, 0.05), pan=0.15)
            P.add(t, coro(nota(acorde[0] + 24), BAR + 1.0, 0.05), pan=-0.2)
            P.add(t, coro(nota(acorde[2] + 12), BAR + 1.0, 0.045), pan=0.25)
            for j, golpe in enumerate([0, 1.5, 2, 3, 3.5]):
                P.add(t + golpe * BEAT, taiko(0.3 if golpe in (0, 2) else 0.18), pan=(-0.2 if j % 2 else 0.2))
            P.add(t, timbal(0.24, 62 if raiz % 12 == 2 else 58))
        if nombre == "crece" and k >= 8:
            # redoble que anuncia la parte epica
            for j in range(int((k - 7) * 4)):
                P.add(t + j * BAR / ((k - 7) * 4), taiko(0.08 + 0.03 * (k - 7)), pan=0.0)
        if nombre == "final":
            P.add(t, bajo(nota(raiz - 12), BAR * 0.95, 0.14))
            if k == 6:
                P.add(t, timbal(0.5, 62)); P.add(t + BEAT * 2, timbal(0.4, 62))
        bar += 1

# melodias: tierna en la intro (campanitas), en la parte epica (metales + cuerdas en octava)
t_crece = secciones[0][1] * BAR
t_epica = (secciones[0][1] + secciones[1][1]) * BAR
t_final = t_epica + secciones[2][1] * BAR
tocar_melodia(BAR * 1, lambda f, d: campana(f, 3.0, 0.19), -12, pan=0.1)
tocar_melodia(t_crece + BAR * 4, lambda f, d: cuerda_corta(f, d * 0.9, 0.12), 0, pan=0.0)
tocar_melodia(t_epica, lambda f, d: metal(f, d, 0.1), 0, pan=0.0)
tocar_melodia(t_epica, lambda f, d: cuerda_corta(f, d * 0.9, 0.08), 12, pan=0.3)
tocar_melodia(t_epica + BAR * 8, lambda f, d: coro(f, d + 0.5, 0.06), 0, pan=-0.3)
tocar_melodia(t_epica + BAR * 14, lambda f, d: metal(f, d, 0.11), 12, pan=0.0)
P.add(t_epica - 3.0, platillo(3.0, 0.1, True))
P.add(t_epica, platillo(3.5, 0.08, False))
P.add(t_final - 3.0, platillo(3.0, 0.08, True))
# acorde final largo en re mayor
for m in (D_ + [62, 66, 69, 74]):
    P.add(t_final + 6 * BAR, pad(nota(m + 12), 9.0, 2000, 0.09))
    P.add(t_final + 6 * BAR, coro(nota(m + 12), 9.0, 0.03))

# ------------------------------------------------------------------ reverb (convolucion con ruido que decae)
def reverb(x, segs=2.8, mezcla=0.32, semilla=1):
    r = np.random.default_rng(semilla)
    n = int(segs * SR); t = np.arange(n) / SR
    ir = r.standard_normal(n) * np.exp(-t * 3.0 / segs * 2.3)
    ir = lp_fijo(ir, 5000); ir /= np.sqrt(np.sum(ir ** 2))
    N = 1 << int(np.ceil(np.log2(len(x) + n)))
    y = np.fft.irfft(np.fft.rfft(x, N) * np.fft.rfft(ir, N), N)[:len(x)]
    return x * (1 - mezcla) + y * mezcla * 1.6

def guardar(nombre, L, R):
    m = max(np.max(np.abs(L)), np.max(np.abs(R)), 1e-6)
    L = np.tanh(L / m * 1.15) * 0.92; R = np.tanh(R / m * 1.15) * 0.92
    datos = (np.stack([L, R], 1) * 32767).astype(np.int16)
    with wave.open(os.path.join(OUT, nombre), "wb") as w:
        w.setnchannels(2); w.setsampwidth(2); w.setframerate(SR); w.writeframes(datos.tobytes())
    print(nombre, round(len(L) / SR, 1), "s")

L = reverb(hp_fijo(P.L, 45), 3.0, 0.35, 1); R = reverb(hp_fijo(P.R, 45), 3.0, 0.35, 2)
# fundido de salida
fade = np.ones(len(L)); nf = int(6 * SR); fade[-nf:] = np.linspace(1, 0, nf) ** 2
guardar("Final_Musica.wav", L * fade, R * fade)

# ------------------------------------------------------------------ subida para el crecimiento
dur = 9.0; n = int(dur * SR); t = np.arange(n) / SR
pico = 6.2
sub = np.clip(t / pico, 0, 1)
ruido = rng.standard_normal(n)
barrido = hp_fijo(ruido, 300) * 0.0
# barrido de ruido: se aclara al subir (mezcla de dos filtros)
grave = lp_fijo(ruido, 600); agudo = lp_fijo(ruido, 7000)
barrido = (grave * (1 - sub) + agudo * sub) * (sub ** 2) * (t < pico) * 0.5
gl = np.zeros(n)
for k, a in ((1, 1.0), (1.5, 0.5), (2, 0.4), (3, 0.2)):
    f = (180 + 1400 * sub ** 2) * k
    gl += a * np.sin(2 * np.pi * np.cumsum(f) / SR)
gl *= (sub ** 1.5) * (t < pico) * 0.3
brillo = sum(np.sin(2 * np.pi * (2000 + 600 * i) * t + i) * (0.5 + 0.5 * np.sin(2 * np.pi * (7 + i) * t)) for i in range(4)) * (sub ** 3) * (t < pico) * 0.05
tg = np.clip(t - pico, 0, None)
golpe = (np.sin(2 * np.pi * np.cumsum(38 * (1 + 2 * np.exp(-tg * 12))) / SR) * np.exp(-tg * 1.2) + lp_fijo(ruido, 1500) * np.exp(-tg * 4) * 0.8) * (t >= pico)
x = barrido + gl + brillo + golpe * 1.3
L = reverb(x, 3.5, 0.4, 3); R = reverb(x, 3.5, 0.4, 4)
guardar("Final_Crecer.wav", L, R)
