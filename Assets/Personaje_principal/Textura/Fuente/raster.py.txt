# z-buffer de la malla alta vista de frente, en coordenadas del dibujo (U,V). Guarda id de triangulo y sombreado.
import sys, time; sys.path.insert(0,'.')
import numpy as np
from comun import cargar
from PIL import Image, ImageDraw
t0=time.time()
P,N,T = cargar(); UV=np.load("uv_imagen.npy"); U,V=UV[:,0],UV[:,1]
H,W=1024,848; K=2                      # supermuestreo x2
X=U*K; Y=V*K; Z=P[:,1]                 # frente = -y (menor y = mas cerca)
x0=X[T]; y0=Y[T]; z0=Z[T]
# solo triangulos que miran al frente o de canto (los de atras nunca se ven)
area=(x0[:,1]-x0[:,0])*(y0[:,2]-y0[:,0])-(x0[:,2]-x0[:,0])*(y0[:,1]-y0[:,0])
zb=np.full((H*K,W*K),np.inf); tid=np.full((H*K,W*K),-1,np.int32)
xmin=np.floor(x0.min(1)).astype(int); xmax=np.ceil(x0.max(1)).astype(int)
ymin=np.floor(y0.min(1)).astype(int); ymax=np.ceil(y0.max(1)).astype(int)
# triangulos pequeños: casi todos caben en 1-3 px; agrupados por tamaño de caja
tam=np.maximum(xmax-xmin,ymax-ymin)+1
for s in np.unique(tam):
    ids=np.where(tam==s)[0]
    if len(ids)==0: continue
    for i0 in range(0,len(ids),200000):
        I=ids[i0:i0+200000]
        oy,ox=np.mgrid[0:s,0:s]; ox=ox.ravel(); oy=oy.ravel()
        px=xmin[I][:,None]+ox[None]; py=ymin[I][:,None]+oy[None]
        cx=px+0.5; cy=py+0.5
        ax,ay=x0[I,0][:,None],y0[I,0][:,None]; bx,by=x0[I,1][:,None],y0[I,1][:,None]; qx,qy=x0[I,2][:,None],y0[I,2][:,None]
        d=(by-qy)*(ax-qx)+(qx-bx)*(ay-qy); d=np.where(np.abs(d)<1e-9,1e-9,d)
        w0=((by-qy)*(cx-qx)+(qx-bx)*(cy-qy))/d; w1=((qy-ay)*(cx-qx)+(ax-qx)*(cy-qy))/d; w2=1-w0-w1
        dentro=(w0>=-1e-4)&(w1>=-1e-4)&(w2>=-1e-4)&(px>=0)&(px<W*K)&(py>=0)&(py<H*K)
        z=w0*z0[I,0][:,None]+w1*z0[I,1][:,None]+w2*z0[I,2][:,None]
        tt=np.broadcast_to(I[:,None],px.shape)
        px=px[dentro]; py=py[dentro]; z=z[dentro]; tt=tt[dentro]
        # menor z gana: ordenar por z descendente y escribir (el ultimo gana)
        o=np.argsort(-z); lin=py[o]*W*K+px[o]
        zf=zb.ravel(); tf=tid.ravel()
        # escribir con minimo real
        np.minimum.at(zf,lin,z[o])
        gana = z[o] <= zf[lin]
        tf[lin[gana]]=tt[o][gana]
print("raster", round(time.time()-t0,1),"s")
np.save("ztid.npy", tid); np.save("zbuf.npy", zb)
# sombreado plano por triangulo
Nt=np.cross(P[T[:,1]]-P[T[:,0]],P[T[:,2]]-P[T[:,0]]); Nt/=np.linalg.norm(Nt,axis=1,keepdims=True)+1e-12
luz=np.array([-0.35,-0.8,0.5]); luz/=np.linalg.norm(luz)
sh=np.abs(Nt@luz)*0.75+0.25
img=np.where(tid>=0, sh[np.maximum(tid,0)], 0.0)
im=Image.fromarray((img*255).astype(np.uint8)).resize((W,H),Image.LANCZOS)
im.save("frente_limpio.png")
np.save("frente_limpio.npy", np.asarray(im).astype(np.float32)/255)
