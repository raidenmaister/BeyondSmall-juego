import numpy as np
from PIL import Image
def cargar():
    nv, nt = np.fromfile("malla.bin", dtype=np.int32, count=2)
    d = np.fromfile("malla.bin", dtype=np.float32, offset=8, count=nv*6)
    P = d[:nv*3].reshape(-1,3).astype(np.float64); N = d[nv*3:].reshape(-1,3).astype(np.float64)
    T = np.fromfile("malla.bin", dtype=np.int32, offset=8+nv*24, count=nt*3).reshape(-1,3)
    return P, N, T
def imagen():
    im = np.asarray(Image.open("referencia.jpg").convert("RGB")).astype(np.float32)/255.
    bg = np.array([179,179,179])/255.
    mask = np.abs(im-bg).max(axis=2) > 18/255.
    return im, mask
def silueta(P, sx, sy, tx, ty, w=848, h=1024):
    u = np.round(tx + P[:,0]*sx).astype(int); v = np.round(ty - P[:,2]*sy).astype(int)
    ok = (u>=0)&(u<w)&(v>=0)&(v<h)
    m = np.zeros((h,w), bool); m[v[ok],u[ok]] = True
    return m
def dil(m,k=2):
    out = m.copy()
    for dy in range(-k,k+1):
        for dx in range(-k,k+1): out |= np.roll(np.roll(m,dy,0),dx,1)
    return out
